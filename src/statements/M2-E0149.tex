\begin{enumerate}[label=(\alph*),ref=\alph*,leftmargin=*]
\item $(x^2-xy-y^2)\,dx-xy\,dy=0$.
\item $(2x+y+1)\,dx+(4x+2y+3)\,dy=0$.
\end{enumerate}
