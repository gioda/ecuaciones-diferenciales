\noindent\textbf{(b)} \[x(t)=\frac{F_0}{k-m\omega^2}\left[\sin\omega t-\frac{\omega}{\omega_0}\sin\omega_0t\right].\]

\noindent\textbf{(c)} \[x(t)=\frac{F_0}{2m\omega_0^2}[\sin\omega_0t-\omega_0t\cos\omega_0t].\]
